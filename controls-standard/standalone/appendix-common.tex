% ============================================================
%  Shared setup for the standalone appendix builds.
%  Each standalone/appendix-*.tex file defines \EFCappkey and
%  \EFCapptitle, then loads this file. build-all.bat runs them.
% ============================================================
\makeatletter
% References: labels inside this appendix resolve normally; labels that
% live in the main document (imported from build/master-refs.aux with the
% prefix M-) print their number followed by "of the Controls Standard".
\newcommand{\EFC@setupref}{%
  \NewCommandCopy\EFC@origref\ref
  \DeclareRobustCommand{\ref}[1]{%
    \@ifundefined{r@##1}{\EFC@origref*{M-##1} of the \documenttitle}{\EFC@origref{##1}}}%
}
\AtBeginDocument{\EFC@setupref}

% Start the appendix with the same letter it has in the full document.
\newcommand{\EFCstartappendix}{%
  \appendix
  \setcounter{chapter}{\csname appnum@\EFCappkey\endcsname}%
}

% Footer identifies the standalone copy.
\fancyfoot[L]{\scriptsize\documenttitle\ rev-\documentrevision\ --- Appendix~\thechapter}
\fancyfoot[R]{\scriptsize\documentdate}
\fancypagestyle{plain}{\fancyhf{}\fancyfoot[C]{\thepage}%
  \fancyfoot[L]{\scriptsize\documenttitle\ rev-\documentrevision\ --- Appendix~\thechapter}%
  \fancyfoot[R]{\scriptsize\documentdate}\renewcommand{\headrulewidth}{0pt}}

% Note under the appendix heading.
\titleformat{\chapter}[hang]{\normalfont\Large\bfseries\color{headblue}}{\thechapter.}{0.5em}{}%
  [\vspace{2pt}{\normalfont\small\color{black!70}Appendix~\thechapter\ of the \companyname\ \documenttitle, Revision~\documentrevision, \documentdate. Standalone copy; see the full document for the other chapters and appendices.}]

% Tell build-all.bat what to call the finished PDF.
\newwrite\EFC@namefile
\AtEndDocument{%
  \immediate\openout\EFC@namefile=\jobname.name
  \immediate\write\EFC@namefile{\documenttitle\space rev-\documentrevision\space - Appendix \thechapter\space - \EFCapptitle}%
  \immediate\closeout\EFC@namefile}
\makeatother
